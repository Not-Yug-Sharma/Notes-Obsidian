// Program  to demonstrate the use of Bitwise Operators

#include<stdio.h>

int main(){
    int a = 5;      // 0101
    int b = 3;      // 0011
    printf("a = %d, b = %d\n", a, b);
    printf("Bitwise AND (a & b) = %d\n", a & b);
    printf("Bitwise OR (a | b) = %d\n", a | b);
    printf("Bitwise XOR (a ^ b) = %d\n", a ^ b);
    printf("Bitwise NOT (~a) = %d\n", ~a);
    printf("Left Shift (a << 1) = %d\n", a << 1);
    printf("Right Shift (a >> 1) = %d\n", a >> 1);
    return 0;

}

// Program to calculate marks percentage and division

#include<stdio.h>

int main(){
    float marks[5], total = 0;
    float percentage;
    printf("Enter marks obtained in 5 subjects: ");
    for (int i = 0; i < 5; i++){
        scanf("%f", &marks[i]);
        total += marks[i];//basically total = total + marks[i];
    }
    percentage = (total / 500.0) * 100;
    printf("Total Marks: %f/500\n", total);
    printf("Percentage: %.2f%%\n", percentage);
    if (percentage >= 80){
        printf("Division: Distinction\n");
    }
    else if (percentage >= 70){
        printf("Division: First Division\n");
    }
    else if (percentage >= 60){
        printf("Division: Second Division\n");
    }
    else if (percentage >= 50){
        printf("Division: Third Division\n");
    }
    else{
        printf("Result: Fail\n");
    }
    return 0;
}

// Program to calcualte factorial

#include<stdio.h>

int main(){
    int n;
    double factorial = 1.0;
    printf("Enter a non-negative integer: ");
    scanf("%d", &n);

    if (n < 0){
        printf("Error! Factorial of a negative number doesn't exist.\n");
    }
    else{
        for (int i = 1; i <= n; i++){
            factorial *= i; // factorial = factorial * i;
        }
        printf("Factorial of %d = %.0f\n", n, factorial);
    }
}

// sum of n natural numbers
#include<stdio.h>

int main(){
    int n, sum = 0;
    printf("Enter a positive integer: ");
    scanf("%d", &n);

    for (int i = 1; i <= n; i++){
        sum += i; // sum = sum + i;
    }
    printf("Sum of first %d natural numbers = %d\n", n, sum);
    return 0;
}

// take a number and calculte the sum of its digits
#include<stdio.h>

int main(){
    int n, sum = 0;
    printf("Enter a number: ");
    scanf("%d", &n);
    while(n!=0){
        sum += n % 10; // sum = sum + (n % 10);
        n = n / 10;
    }
    printf("Sum of digits: %d\n", sum);
    return 0;
}

// Program to check if a number is palindrome
#include<stdio.h>

int main(){
    int n, original, reversed = 0;
    printf("Enter an integer: ");
    scanf("%d", &n);
    original = n;

    while (n != 0){
        
        reversed = reversed * 10 +  (n % 10);
        n = n / 10;
    }

    if (original == reversed){
        printf("%d is a palindrome.\n", original);
    }
    else{
        printf("%d is not a palindrome.\n", original);
    }
    return 0;
}
